cmake_minimum_required(VERSION 3.22.1)
project(lx_audio C)
file(GLOB LAME_SOURCES CONFIGURE_DEPENDS lame/libmp3lame/*.c)
add_library(mp3lame SHARED ${LAME_SOURCES})
target_include_directories(mp3lame PRIVATE . lame/include lame/libmp3lame)
target_compile_definitions(mp3lame PRIVATE HAVE_CONFIG_H NDEBUG)
target_compile_options(mp3lame PRIVATE -O2)
target_link_libraries(mp3lame m)
add_library(lxmp3 SHARED mp3_jni.c)
set_target_properties(lxmp3 PROPERTIES C_VISIBILITY_PRESET hidden)
target_include_directories(lxmp3 PRIVATE lame/include)
target_link_libraries(lxmp3 mp3lame)
target_link_options(mp3lame PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
target_link_options(lxmp3 PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
target_link_options(lxmp3 PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
